cmake_minimum_required(VERSION 3.16)
project(CppFundamental LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(BUILD_EXERCISES "Build exercise binaries" ON)
option(BUILD_SOURCE_PROJECTS "Build source binaries" ON)

function(add_lesson_target lesson_dir phase_name topic_name mode)
  file(GLOB LESSON_SOURCES "${lesson_dir}/${mode}/*.cpp")
  if(LESSON_SOURCES)
    string(REPLACE "-" "_" target_name "${phase_name}_${topic_name}_${mode}")
    string(REPLACE " " "_" target_name "${target_name}")
    add_executable(${target_name} ${LESSON_SOURCES})
    target_include_directories(${target_name} PRIVATE "${lesson_dir}/${mode}")
  endif()
endfunction()

file(GLOB PHASE_DIRS RELATIVE "${CMAKE_SOURCE_DIR}" "${CMAKE_SOURCE_DIR}/Phase_*")
foreach(phase_dir IN LISTS PHASE_DIRS)
  if(IS_DIRECTORY "${CMAKE_SOURCE_DIR}/${phase_dir}")
    get_filename_component(phase_name "${phase_dir}" NAME)
    file(GLOB TOPIC_DIRS RELATIVE "${CMAKE_SOURCE_DIR}/${phase_dir}" "${CMAKE_SOURCE_DIR}/${phase_dir}/*")
    foreach(topic_rel IN LISTS TOPIC_DIRS)
      set(topic_dir "${CMAKE_SOURCE_DIR}/${phase_dir}/${topic_rel}")
      if(IS_DIRECTORY "${topic_dir}")
        get_filename_component(topic_name "${topic_rel}" NAME)
        if(BUILD_EXERCISES)
          add_lesson_target("${topic_dir}" "${phase_name}" "${topic_name}" exercise)
        endif()
        if(BUILD_SOURCE_PROJECTS)
          add_lesson_target("${topic_dir}" "${phase_name}" "${topic_name}" source)
        endif()
      endif()
    endforeach()
  endif()
endforeach()
